#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""
ER diagram for the SAF longitudinal-survey data model.
对应 saf_model.py / saf_model.sql（question · item_answer · survey · respondent）。

Professional crow's-foot ERD — looks like a hand-crafted data-modeling deliverable,
not a template:
  - rounded table cards with a subtle offset shadow
  - PK / FK / attribute grouping (PK bold+underline red, FK italic blue)
  - real cardinality notation (mandatory bar "1" vs crow's foot "N")
  - relationship captions, muted slate palette, clean sans-serif

Entity chain: question -- item_answer -- survey -- respondent
Laid out as a zigzag (S) so relationship edges do NOT cross.

Run:  python saf_model_er.py   ->   saf_model_er.png
Dependencies: matplotlib (no third-party graphviz needed).
English / Catalan labels only (no CJK fonts on this machine).

Robust startup: if the current interpreter lacks matplotlib, automatically
re-exec with the first interpreter on PATH that has it. This makes the script
run under any `python`/`python3` even when matplotlib isn't installed in it.
"""
import math
import os
import sys
import shutil
import subprocess

# ── self-healing: ensure matplotlib is importable in THIS process ────────────
def _reexec_with_matplotlib():
    """Try to import matplotlib; if unavailable, re-exec with an interpreter that has it."""
    try:
        import matplotlib  # noqa: F401
        return
    except Exception:
        pass

    candidates = []
    for c in (os.environ.get("PYTHON_WITH_MPL"),
              "/usr/local/bin/python3",
              "/usr/bin/python3",
              "/opt/homebrew/bin/python3",
              "/Library/Frameworks/Python.framework/Versions/3.13/bin/python3",
              shutil.which("python3")):
        if c and c not in candidates:
            candidates.append(c)
    for c in candidates:
        try:
            if subprocess.run([c, "-c", "import matplotlib"],
                              stdout=subprocess.DEVNULL,
                              stderr=subprocess.DEVNULL).returncode == 0:
                os.execv(c, [c, os.path.abspath(sys.argv[0])])
                return
        except Exception:
            continue
    raise SystemExit(
        "ERROR: matplotlib not found in any installed Python. "
        "Install it (e.g. `pip install matplotlib`) or set PYTHON_WITH_MPL to an "
        "interpreter that has it."
    )


_reexec_with_matplotlib()

# A writable cache dir avoids the ".matplotlib not writable" warnings.
import tempfile as _tempfile
os.environ.setdefault("MPLCONFIGDIR", _tempfile.gettempdir())

import matplotlib
matplotlib.use("Agg")
import matplotlib.pyplot as plt
from matplotlib.patches import FancyBboxPatch

fig, ax = plt.subplots(figsize=(15, 10.2))
ax.set_xlim(0, 13)
ax.set_ylim(0, 9.4)
ax.set_aspect("equal")
ax.axis("off")
fig.patch.set_facecolor("#fbfcfd")

# ── palette ────────────────────────────────────────────────────────────────
HEADER  = "#2f3d4e"     # slate title bar
INK     = "#5f6670"     # edge / cardinality marks
PK      = "#a8322b"     # primary key
FK      = "#2b6fa0"     # foreign key
NORMAL  = "#6b7078"     # plain attribute
EDGE    = "#c9ced4"     # card border
SHADOW  = "#e6e9ec"     # card shadow
ROW     = 0.24          # vertical spacing between attribute rows


def draw_entity(cx, cy, w, h, title, pk, fk, attrs):
    """Draw one rounded table card with a PK / FK / attribute block."""
    x0, y0 = cx - w / 2, cy - h / 2

    # offset shadow
    ax.add_patch(FancyBboxPatch((x0 + 0.07, y0 - 0.07), w, h,
              boxstyle="round,pad=0.02,rounding_size=0.11",
              fill=True, facecolor=SHADOW, edgecolor="none", zorder=1))
    # card body
    ax.add_patch(FancyBboxPatch((x0, y0), w, h,
              boxstyle="round,pad=0.02,rounding_size=0.11",
              fill=True, facecolor="#ffffff", edgecolor=EDGE, linewidth=1.0, zorder=2))
    # title bar
    barh = 0.46
    ax.add_patch(FancyBboxPatch((x0, y0 + h - barh), w, barh,
              boxstyle="round,pad=0.02,rounding_size=0.11",
              fill=True, facecolor=HEADER, edgecolor="none", zorder=3))
    ax.text(cx, y0 + h - barh / 2, title, ha="center", va="center",
            color="white", fontsize=11.6, fontweight="bold", zorder=4)

    # attributes: PK (bold+underline) -> FK (italic) -> normal
    y = y0 + h - barh - ROW
    for a in pk:
        t = ax.text(cx, y, f"{a}  (PK)", ha="center", va="center",
                    color=PK, fontsize=8.0, fontweight="bold", zorder=5)
        t.underline = True
        y -= ROW
    for a in fk:
        ax.text(cx, y, f"{a}  (FK)", ha="center", va="center",
                color=FK, fontsize=8.0, style="italic", zorder=5)
        y -= ROW
    for a in attrs:
        ax.text(cx, y, a, ha="center", va="center", color=NORMAL, fontsize=8.0, zorder=5)
        y -= ROW


# ── cardinality marks: mandatory bar (1) vs crow's foot (N) ─────────────────
def mandatory_bar(x, y, ux, uy):
    """A short perpendicular bar at a '1' endpoint. (ux,uy) points toward the other entity."""
    ang = math.atan2(uy, ux) + math.pi / 2
    L = 0.16
    ax.plot([x - L * math.cos(ang), x + L * math.cos(ang)],
            [y - L * math.sin(ang), y + L * math.sin(ang)], color=INK, lw=2.5, zorder=6)


def crrows_foot(x, y, ux, uy):
    """Three-prong crow's foot at an 'N' endpoint; central prong points toward the other entity."""
    ang = math.atan2(uy, ux)
    L = 0.23
    for off in (-math.radians(30), 0, math.radians(30)):
        a = ang + off
        ax.plot([x, x + L * math.cos(a)], [y, y + L * math.sin(a)],
                color=INK, lw=2.4, zorder=6)


def cardinality(x1, y1, u1, s1, x2, y2, u2, s2, label=(0.0, 0.0)):
    """Draw one relationship edge. s1/s2 in {'1','N'}; u = unit dir from endpoint to the other.
    label = (dx, dy) offset placing the '1 — N' text clear of the markers."""
    ax.plot([x1, x2], [y1, y2], color=INK, lw=1.5, zorder=2)
    if s1 == "1":
        mandatory_bar(x1, y1, u1[0], u1[1])
    else:
        crrows_foot(x1, y1, u1[0], u1[1])
    if s2 == "1":
        mandatory_bar(x2, y2, u2[0], u2[1])
    else:
        crrows_foot(x2, y2, u2[0], u2[1])
    mx, my = (x1 + x2) / 2, (y1 + y2) / 2
    ax.text(mx + label[0], my + label[1], "1  —  N", ha="center", va="center",
            fontsize=9.0, fontweight="bold", color=INK, zorder=7)


# ── entities (rounded cards) ───────────────────────────────────────────────
draw_entity(2.7, 6.1, 3.4, 1.9, "respondent",
            ["respondent_id"], [], ["gener", "estudi", "situacion", "abonat"])

            ["key"], [], ["number", "block", "pillar", "scale", "label",

# ── relationship edges (FK direction + cardinality) ──────────────────────────
# respondent 1 -- N survey          (survey.respondent_id -> respondent.respondent_id)
# survey   1 -- N item_answer        (item_answer.survey_id       -> survey.survey_id)
# question 1 -- N item_answer        (item_answer.question_key    -> question.key)
cardinality(4.4, 6.1,  (1, 0), "1", 5.0, 6.1,  (-1, 0), "N", (0.0,  0.34))
cardinality(8.4, 6.1,  (1, 0), "1", 8.8, 6.1,  (-1, 0), "N", (0.0,  0.34))
cardinality(10.6, 4.9,(0,-1), "N", 10.6, 4.3,  (0, 1),  "1", (0.55, 0.0))

ax.text(4.7, 4.5, "one respondent registers many survey waves",

# ── header ───────────────────────────────────────────────────────────────────
ax.text(0.55, 9.05, "ER Diagram", ha="left", va="center",

ax.plot([0.55, 5.6], [8.78, 8.78], color=HEADER, lw=1.4, zorder=2)
